// Old code: only checked for 60 seconds (20 attempts × 3s), then gave up. // If the person paid and came back after 60s, the page was frozen on "pending" // even though Firestore already had status: 'approved'. // // New code: // 1. Checks IMMEDIATELY on page load (before starting the interval) // → person pays, comes back 5 minutes later, page loads → instant redirect // 2. Checks every 5 seconds for up to 40 attempts = 3.5 minutes of polling // 3. Re-checks on window focus (person switches tabs, comes back → checks again) // 4. After polling stops, "Verificar Novamente" button does a fresh immediate check // instead of just reloading the page // ──────────────────────────────────────────────────────────────────────────── const MAX_ATTEMPTS = 40; // 40 × 5s = 3.5 minutes of polling const POLL_INTERVAL_MS = 5000; // check every 5 seconds function getUrlParams() { const params = {}; const urlParams = new URLSearchParams(window.location.search); for (const [key, value] of urlParams.entries()) { params[key] = value; } return params; } function getExternalReference() { const params = getUrlParams(); let externalReference = params.external_reference; if (!externalReference) { const lastPayment = sessionStorage.getItem('lastPayment'); if (lastPayment) { try { const paymentData = JSON.parse(lastPayment); externalReference = paymentData.external_reference; } catch(e) {} } } return externalReference; } // Single status check — returns true if terminal (approved/failed), false if still pending async function checkOnce(externalReference) { const response = await fetch( `${FUNCTIONS_BASE_URL}/getPaymentStatus?external_reference=${externalReference}` ); if (!response.ok) return false; const data = await response.json(); if (!data.success || !data.payment) return false; const payment = data.payment; // Update amount display document.getElementById('orderAmount').textContent = new Intl.NumberFormat('pt-BR', { style: 'currency', currency: 'BRL' }).format(payment.total_amount || 0); const statusBadge = document.getElementById('paymentStatus'); if (payment.status === 'approved') { console.log('✅ PAYMENT APPROVED — redirecting'); statusBadge.textContent = 'APROVADO!'; statusBadge.className = 'status-badge status-approved'; document.getElementById('spinner').style.display = 'none'; document.getElementById('statusMessage').textContent = 'Pagamento confirmado! Redirecionando...'; document.querySelector('.payment-icon').textContent = '✅'; stopPolling(); setTimeout(() => { window.location.href = `/success.html?external_reference=${externalReference}&status=approved`; }, 1500); return true; // terminal } else if ( payment.status === 'rejected' || payment.status === 'cancelled' || payment.status === 'failure' || payment.status === 'refunded' ) { stopPolling(); showFailedPayment('Pagamento não foi aprovado'); return true; // terminal } else { // Still pending — update badge statusBadge.textContent = 'Aguardando pagamento'; statusBadge.className = 'status-badge status-pending'; if (payment.payment_details?.method === 'pix' || payment.payment_details?.type === 'bank_transfer') { document.getElementById('pixInstructions').classList.remove('d-none'); } else if (payment.payment_details?.method === 'bolbradesco') { document.getElementById('boletoInstructions').classList.remove('d-none'); } return false; // not terminal, keep polling } } function stopPolling() { if (checkInterval) { clearInterval(checkInterval); checkInterval = null; } } async function startChecking() { const params = getUrlParams(); const externalReference = getExternalReference(); document.getElementById('orderId').textContent = externalReference || 'N/A'; document.getElementById('maxAttempts').textContent = MAX_ATTEMPTS; const paymentType = params.payment_type; if (paymentType === 'bank_transfer' || paymentType === 'pix') { document.getElementById('pixInstructions').classList.remove('d-none'); } else if (paymentType === 'ticket') { document.getElementById('boletoInstructions').classList.remove('d-none'); } // If URL already says approved/rejected, handle immediately const collectionStatus = params.collection_status; if (collectionStatus === 'approved' || collectionStatus === 'success') { window.location.href = `/success.html?external_reference=${externalReference}&status=approved`; return; } else if (collectionStatus === 'rejected' || collectionStatus === 'failure') { showFailedPayment('Pagamento rejeitado'); return; } if (!externalReference) { showError('Nenhuma referência de pedido encontrada'); return; } // ─── CHECK IMMEDIATELY ON PAGE LOAD ───────────────────────────────── // This is the main fix: if the person already paid and is now loading // this page, we catch it instantly without waiting for the first interval tick console.log('🔍 Immediate check on page load...'); try { const done = await checkOnce(externalReference); if (done) return; // already approved or failed — stop here } catch(e) { console.warn('Immediate check failed:', e.message); } // ─── THEN POLL ON INTERVAL ─────────────────────────────────────────── checkInterval = setInterval(async () => { attempts++; document.getElementById('attemptCount').textContent = attempts; console.log(`🔄 Polling attempt ${attempts}/${MAX_ATTEMPTS}...`); try { const done = await checkOnce(externalReference); if (done) return; } catch (error) { console.error('❌ Poll error:', error); } if (attempts >= MAX_ATTEMPTS) { stopPolling(); document.getElementById('spinner').style.display = 'none'; document.getElementById('statusMessage').textContent = 'Ainda aguardamos confirmação. Use o botão abaixo para verificar manualmente.'; } }, POLL_INTERVAL_MS); } function showFailedPayment(message) { stopPolling(); document.getElementById('spinner').style.display = 'none'; document.getElementById('statusMessage').textContent = message; document.getElementById('paymentStatus').textContent = 'FALHOU'; document.getElementById('paymentStatus').className = 'status-badge status-failed'; document.querySelector('.payment-icon').textContent = '❌'; document.querySelector('h2').textContent = 'Pagamento Não Concluído'; const btn = document.getElementById('checkAgainBtn'); btn.innerHTML = ' Tentar Novamente'; btn.onclick = () => { window.location.href = '/checkout.html'; }; } function showError(message) { stopPolling(); document.getElementById('spinner').style.display = 'none'; document.getElementById('statusMessage').textContent = message; document.getElementById('paymentStatus').textContent = 'ERRO'; document.getElementById('paymentStatus').className = 'status-badge status-failed'; } // Manual check button — does a fresh immediate check instead of reloading document.getElementById('checkAgainBtn').addEventListener('click', async function() { const externalReference = getExternalReference(); if (!externalReference) { window.location.reload(); return; } this.disabled = true; this.textContent = 'Verificando...'; try { await checkOnce(externalReference); } catch(e) { console.error('Manual check failed:', e); } this.disabled = false; this.innerHTML = ' Verificar Novamente'; }); // Re-check when person switches back to this tab // (e.g. paid on mobile banking app, switches back to browser) window.addEventListener('focus', async function() { const externalReference = getExternalReference(); if (!externalReference || !checkInterval) return; // only while polling is active console.log('👁️ Page refocused — checking immediately'); try { await checkOnce(externalReference); } catch(e) {} }); // Start everything when page loads document.addEventListener('DOMContentLoaded', startChecking);
